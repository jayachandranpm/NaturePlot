Plant your first chart

Read the full API reference
your-first-garden.js
// A month of small steps.
import { NaturePlot } from 'natureplot';

const garden = new NaturePlot('#chart', {
  type: 'garden',
  theme: 'meadow',
  startDate: '2026-09-01',
  days: 30,
  data: [
    { date: '2026-09-01', value: 8 },
    { date: '2026-09-02', value: 12 },
    { date: '2026-09-03', value: 6 },
  ],
});

// Let it change with the seasons.
garden.setTheme('autumn');
